# while True:
#     try:
#         # 1. If heap empty: await wake_event; clear-before-work; continue.
#         # 2. Peek front tuple; identify item.
#         # 3. Preemption accounting:
#         #    - If front item differs from self._current_front:

#         #      * Set new front's current_front_since = now; self._current_front = front_item.

#         #      (total_front_time + (now - current_front_since)) >= timing.max_hold_seconds.
#         # 5. If eligible: heappop; clear current_front_since; await bus.dispatch(BufferedDelivery(prompt=item.prompt, items=[item])); self._current_front = None; continue.

#         #      idle_window target - (now - last_message_time),

#         # 8. await asyncio.wait({wake_task, timer_task} minus None, return_when=FIRST_COMPLETED).
#         # 9. Cancel the loser and await it to drain CancelledError (per research pointer).
