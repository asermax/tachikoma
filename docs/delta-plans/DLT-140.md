  def validate_file_path(
      file_path: str,
      workspace_path: Path,
      allowed_roots: tuple[Path, ...],
  ) -> Path:
      # 1. build path; if not absolute, prefix with workspace_path (note that
      #    workspace_path is passed separately — see S1 in the design: it anchors

      # 2. resolved = path.resolve()  (symlinks collapsed; relies on file existing

#### Step 7: Factory computes + captures `allowed_roots`

- Create/modify: `src/tachikoma/telegram/tools.py`
- What to do: Extend `create_send_file_server` with a new `configured_extra_roots: list[Path]` parameter. Compute the resolved, dedup'd tuple once at the top of the factory and capture it in the inner closure so the tool call reuses it. Update the `@tool(...)` closure to pass the tuple through. Import `tempfile` at the top of the module.

  ```python
  import tempfile
  from pathlib import Path
  # ... other imports unchanged ...

  def create_send_file_server(
      bot: Bot,
      chat_id: int,
      workspace_path: Path,
      configured_extra_roots: list[Path],
  ) -> McpSdkServerConfig:
      # Compute once at factory call time (see design §Key Decisions → "Resolve
      # allowed roots once at factory-call time"); dict.fromkeys collapses duplicates
      allowed_roots: tuple[Path, ...] = tuple(
          dict.fromkeys(
              p.resolve()
              for p in (workspace_path, Path(tempfile.gettempdir()), *configured_extra_roots)
          )
      )

      @tool("send_file", <updated description — see Step 8>, SendFileArgs.model_json_schema())
      async def send_file(args: dict) -> dict:
          parsed = SendFileArgs.model_validate(args)
          return await handle_send_file(
              parsed.file_path,
              parsed.caption,
              bot,
              chat_id,
              workspace_path,
              allowed_roots,
          )

      return create_sdk_mcp_server(name="send-file", tools=[send_file])
  ```
